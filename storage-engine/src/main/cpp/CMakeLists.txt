cmake_minimum_required(VERSION 3.22)
project(infiniext C)
file(GLOB EXT_SOURCES CONFIGURE_DEPENDS third_party/lwext4/src/*.c)
list(FILTER EXT_SOURCES EXCLUDE REGEX "ext4_(mkfs|mbr)\\.c$")
add_library(infiniext SHARED ext_backend.c ext_jni.c ext_metadata.c ${EXT_SOURCES})
target_include_directories(infiniext PRIVATE third_party/lwext4/include)
target_compile_definitions(infiniext PRIVATE CONFIG_USE_DEFAULT_CFG=1 CONFIG_DEBUG_PRINTF=0 CONFIG_DEBUG_ASSERT=0 CONFIG_BLOCK_DEV_CACHE_SIZE=64)
target_compile_options(infiniext PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_options(infiniext PRIVATE -Wl,-z,max-page-size=16384)
